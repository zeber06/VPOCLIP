# Szablony nazw sa uzywane tylko przy Process_All_Layouts=YES.
# Dostepne tokeny:
#   <source> - nazwa pliku zrodlowego bez rozszerzenia
#   <layout> - nazwa eksportowanego layoutu
# Szablon aktywnego formatu musi zawierac token <layout>.
# Przyklady:
#   <source>_<layout>
#   prefix<source>ABCD<layout>sufix
#   prefix<layout>sufix
#   <layout>

# NO: eksportowane sa tylko pozycje z sekcji [TASKS].
# YES: eksportowane sa wszystkie layouty papieru poza Model.
# Przy YES sekcja [TASKS] jest ignorowana.
#
# Wyznaczanie klucza XREF przy Ignore_XREF=NO:
#   Process_All_Layouts=NO  - wzorzec pochodzi z 3. kolumny [TASKS].
#   Process_All_Layouts=YES - wzorzec jest tworzony z 2 pierwszych znakow
#                             nazwy layoutu, np. GH01 -> *- GH.
Process_All_Layouts=NO

# NO: przed wydrukiem PDF ladowane sa tylko XREF-y pasujace do klucza
# zadania, a pozostale sa wyladowywane. Ten sam filtr pozostawia
# odpowiednie XREF-y w wynikowym DWG.
#
# YES: klucz zadania jest ignorowany. Przed PDF ladowane i zachowywane
# sa wszystkie XREF-y. Gdy rysunek zawiera wiecej niz 8 XREF-ow,
# VPOCLIP przed startem poprosi o jednorazowe potwierdzenie.
#
# XREF-y sa przygotowywane przed PDF niezaleznie od ustawienia
# PDF_From_Source. Stan jest wykorzystywany ponownie dla kolejnych zadan
# z tym samym kluczem, aby ograniczyc liczbe operacji Reload/Unload.
Ignore_XREF=NO

# YES: gdy layout nie ma zadnej poprawnej ramki VPOutline, VPOBATCH
# tworzy tymczasowe prostokatne ramki bezposrednio z jego rzutni.
# Ramki i etykiety powstaja tylko w kopii roboczej i sa usuwane przed
# zapisem wyniku. Rysunek zrodlowy nie jest zmieniany.
# Istniejace poprawne ramki zawsze maja pierwszenstwo. Osierocone linki
# VPO_LINK do skasowanych lub nieistniejacych ramek sa usuwane w kopii
# roboczej, a ramki sa tworzone ponownie. Istniejaca, lecz uszkodzona ramka
# nadal zatrzymuje zadanie i nie jest automatycznie naprawiana.
# NO: brak ramek powoduje blad zadania, tak jak w starszych wersjach.
# LCLIP zawsze uzywa automatycznego tworzenia ramek niezaleznie od tej opcji.
Auto_Create_Frames=YES

# Warstwy ignorowane podczas czyszczenia.
# Obiekty na tych warstwach nie sa usuwane ani przycinane poza ramkami.
# Nazwy i wzorce rozdzielaj srednikiem; dozwolone sa symbole * oraz ?.
# Wielkosc liter nie ma znaczenia. Pusta wartosc wylacza dodatkowa ochrone.
# Dla nazw z polskimi znakami uzywaj ? zamiast litery diakrytycznej.
# Przyklad: _Podzia? na odcinki;DEFPOINTS;*OPIS*;XREF_*

# Lista layoutow pomijanych przy Process_All_Layouts=YES.
# Wzorce rozdzielaj srednikiem; dozwolone sa symbole wieloznaczne * i ?.
# W nazwach z polskimi znakami rowniez stosuj ? zamiast litery diakrytycznej.
# Pozostaw pusta wartosc, aby niczego nie wykluczac.
